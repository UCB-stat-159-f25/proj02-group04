Q: How to create clean PCA plot python 
A: import numpy as np
import pandas as pd
from sklearn.decomposition import PCA
import matplotlib.pyplot as plt

def pca_scatter(X, labels=None, n_components=2, title="PCA Scatter Plot"):
    """
    X        : array-like or DataFrame, shape (n_samples, n_features)
    labels   : list/array of point labels (optional)
    n_components : 2 or 3 (for 2D or 3D PCA)
    """
    X = np.asarray(X)
    
    pca = PCA(n_components=n_components)
    coords = pca.fit_transform(X)
    
    explained = pca.explained_variance_ratio_ * 100  # %
    
    if n_components == 2:
        plt.figure(figsize=(8,6))
        plt.scatter(coords[:,0], coords[:,1])
        
        if labels is not None:
            for x, y, lab in zip(coords[:,0], coords[:,1], labels):
                plt.text(x + 0.01, y + 0.01, str(lab), fontsize=8)
        
        plt.xlabel(f"PC1 ({explained[0]:.1f}% var)")
        plt.ylabel(f"PC2 ({explained[1]:.1f}% var)")
        plt.title(title)
        plt.axhline(0, color="gray", linewidth=0.5)
        plt.axvline(0, color="gray", linewidth=0.5)
        plt.show()
    else:
        raise ValueError("Only 2D PCA plot supported in this helper.")
